// src/features/shipments/ShipmentsList.tsx import { useShipments } from '../../hooks/useShipments'; import { ShipmentDetails } from './ShipmentDetails'; import { SensorPanel } from './SensorPanel'; import { AuditTimeline } from './AuditTimeline'; export function ShipmentsList() { const { shipments, loading, error } = useShipments(); return ( <Dashboard> <StatsBar data={shipments} /> <ShipmentTable rows={shipments} /> <SensorPanel shipmentId={selected?.id} /> <AuditTimeline events={selected?.audit} /> </Dashboard> ); }

🗂 Feature-First 目录结构

每个业务域（auth / shipments）独立封装，避免全局耦合，便于团队并行开发。

🔗 自定义 Hooks 抽离逻辑

useShipments / useAuth 等 hooks 将数据获取与 UI 解耦，方便单元测试与逻辑复用。

🔒 路由级鉴权

ProtectedRoute 组件在路由层拦截未授权访问，无需在每个页面重复鉴权逻辑。

🌐 统一 API 层

src/api 目录集中管理所有接口调用，切换后端只需修改一处，前端无感知。

📐 严格 TypeScript 类型

src/types 目录定义全局接口，配合 tsconfig 严格模式，杜绝运行时类型错误。

⚡ Vite 极速开发体验

毫秒级 HMR，构建产物极小，支持 ESM 原生模块，开发体验远超 CRA。
